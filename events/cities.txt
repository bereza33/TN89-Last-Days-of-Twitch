add_namespace = cities

country_event = {
    id = cities.1
    is_triggered_only = yes
    hidden = yes

    immediate = {
        # допустим:
        # 64 = Berlin
        # 1188 = West Berlin

        if = {
            limit = {
                64 = { is_owned_and_controlled_by = ROOT }
                1188 = { is_owned_and_controlled_by = ROOT }
            }

            # unified mode
            64 = { set_state_name = "Вовбург" }
            1188 = { set_state_name = "Западный Вовбург" }

            # убрать VP:
            set_victory_points = { province = 13607 value = 0 }
            
        }

        else = {
            # divided mode
            64 = { set_state_name = "Восточный Вовбург" }
            1188 = { set_state_name = "Западный Вовбург" }

            # вернуть VP:
            set_victory_points = { province = 13607 value = 5 }
        }
    }
}
country_event = {
    id = cities.2
    is_triggered_only = yes
    hidden = yes

    immediate = {
        # допустим:
        # 64 = Berlin
        # 1188 = West Berlin 9523

        if = {
            limit = {
                16 = { is_owned_and_controlled_by = ROOT }
                1032 = { is_owned_and_controlled_by = ROOT }
            }

            # unified mode
            16 = { set_state_name = "Иль де Генсуха" }
            1032 = { set_state_name = "Иль де Генсуха" }

            # убрать VP:
            set_victory_points = { province = 13633 value = 50 }
            set_victory_points = { province = 9523 value = 0 }
            set_capital = {state = 1032 remember_old_capital = yes}
            
        }

        else = {
            # divided mode
            16 = { set_state_name = "Иль де Генсуха (Западный)" }
            1032 = { set_state_name = "Иль де Генсуха (Восточный)" }

            # вернуть VP:
            set_victory_points = { province = 13633 value = 5 }
            set_victory_points = { province = 9523 value = 5 }
        }
    }
}
