demobilization = {

	demobilization_economic_2 = {
		icon = revolt

		allowed = { always = no } #manually activated

		cancel_trigger = {
			OR = {
				has_unsupported_economic_law = no
				has_war = yes
				has_civil_war = yes
			}
		}

		cost = 0
		is_good = no
		selectable_mission = yes
		days_mission_timeout = 70

		complete_effect = {
			if = {
				limit = { low_economic_mobilisation_available = no }
				if = {
					limit = { has_idea = tot_economic_mobilisation }
					add_political_power = 300
					swap_ideas = {
						remove_idea = tot_economic_mobilisation
						add_idea = civilian_economy
					}
				}
				else_if = {
					limit = { has_idea = war_economy }
					add_political_power = 225
					swap_ideas = {
						remove_idea = war_economy
						add_idea = civilian_economy
					}
				}
				else_if = {
					limit = { has_idea = partial_economic_mobilisation }
					add_political_power = 150
					swap_ideas = {
						remove_idea = partial_economic_mobilisation
						add_idea = civilian_economy
					}
				}
				else_if = {
					limit = { has_idea = low_economic_mobilisation }
					add_political_power = 75
					swap_ideas = {
						remove_idea = low_economic_mobilisation
						add_idea = civilian_economy
					}
				}
			}
			else_if = {
				limit = { partial_economic_mobilisation_available = no }
				if = {
					limit = { has_idea = tot_economic_mobilisation }
					add_political_power = 225
					swap_ideas = {
						remove_idea = tot_economic_mobilisation
						add_idea = low_economic_mobilisation
					}
				}
				else_if = {
					limit = { has_idea = war_economy }
					add_political_power = 150
					swap_ideas = {
						remove_idea = war_economy
						add_idea = low_economic_mobilisation
					}
				}
				else_if = {
					limit = { has_idea = partial_economic_mobilisation }
					add_political_power = 75
					swap_ideas = {
						remove_idea = partial_economic_mobilisation
						add_idea = low_economic_mobilisation
					}
				}
			}
			else = {
				if = {
					limit = { has_idea = tot_economic_mobilisation }
					add_political_power = 150
					swap_ideas = {
						remove_idea = tot_economic_mobilisation
						add_idea = partial_economic_mobilisation
					}
				}
				else_if = {
					limit = { has_idea = war_economy }
					add_political_power = 75
					swap_ideas = {
						remove_idea = war_economy
						add_idea = partial_economic_mobilisation
					}
				}
			}
		}

		timeout_effect = {
			activate_mission = demobilization_economic_2
			if = {
				limit = {
					NOT = {
						OR = {
							has_country_flag = crisis_solution_in_progress
							has_stability < 0.5
						}	
					}
				}	
				add_stability = -0.05
			}
			if = {
				limit = {
					has_stability < 0.5
					NOT = { 
						has_country_flag = crisis_solution_in_progress
					}
				}
				random_list = {
					90 = {
						add_stability = -0.05
					}
					10 = {
						modifier = {
							factor = 1.5
							has_stability < 0.21
						}
						modifier = {
							factor = 0.75
							has_stability > 0.5
						}
						modifier = {
							factor = 0.75
							has_stability > 0.7
						}
						modifier = {
							factor = 0.8
							has_government = democratic
						}
						modifier = {
							factor = 0.7
							has_government = communism
						}
						trigger_strikes_escalation = yes
					}
				}
			}
		}	
		fixed_random_seed = no
	}
	
	demob_economic_mission = { #original, disabled
	
		visible = { always = no }
		available = { always = no }
		activation = { always = no }
		days_mission_timeout = 1
		complete_effect = {}
		timeout_effect = {}
	}
	
	demobilization_economic = { #original, disabled

		visible = { always = no }
		available = { always = no }
		activation = { always = no }
		days_mission_timeout = 1
		complete_effect = {}
		timeout_effect = {}
		
	}
}

							has_idea = strikes_1

						}

					}
				}
			}
			NOT = { 
				has_country_flag = crisis_solution_in_progress
			}
		}	

		cost = 0
		ai_will_do = {
			factor = 1
			modifier = {
				has_idea = strikes_3
				factor = 500
			}
		}
		
		visible = {
			OR = {
				has_idea = strikes_1
				has_idea = strikes_2
				has_idea = strikes_3
				has_active_mission = strikes_mission
			}
			NOT = {
				has_country_flag = crisis_solution_in_progress
			}
		}


		complete_effect = {
			country_event = stability.22

	}
}
