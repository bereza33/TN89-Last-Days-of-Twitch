GER_bundestag_support_change = {

	set_temp_variable = {
		var = GON_support_parlament_total_var_temp
		value = 0
	}

	#hoi4
	set_temp_variable = {
		var = GON_hoi4_support_parlament_var_temp
		value = GON_support_hoi4_var
	}
	if = {
		#Indifferent Or Lower
		limit = {
			check_variable = {
				var = GON_support_hoi4_var
				value = 0.6
				compare = less_than
			}
		}
		multiply_temp_variable = {
			var = GON_hoi4_support_parlament_var_temp
			value = 0
		}
	}
	else_if = {
		#Supportive
		limit = {
			check_variable = {
				var = GON_support_hoi4_var
				value = 0.6
				compare = greater_than_or_equals
			}
			check_variable = {
				var = GON_support_hoi4_var
				value = 0.8
				compare = less_than
			}
		}
		multiply_temp_variable = {
			var = GON_hoi4_support_parlament_var_temp
			value = 0.5
		}
	}
	add_to_temp_variable = {
		var = GON_support_parlament_total_var_temp
