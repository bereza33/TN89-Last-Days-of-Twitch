# this creates a pie chart of popularities - can be easily adapted to show anything that adds up to 100%
create_pie_chart = {
	clear_array = pie_chart
	resize_array = {
		array = pie_chart
		value = 1 # 1 is value for neutrality frame, so we don't have to make a loop for it
		size = 100 # 100 pieces, each percent is one piece
	}
	# index pointer
	set_temp_variable = { tmp_pop = stream_gaming }
	multiply_temp_variable = { tmp_pop = 1 }
	round_temp_variable = tmp_pop
	
	### CHATTING ###
	set_temp_variable = { tmp_pop = stream_chatting }
	multiply_temp_variable = { tmp_pop = 1 }
	round_temp_variable = tmp_pop
	set_variable = { ROOT.last_idx = stream_gaming }
	add_to_temp_variable = { tmp_pop = ROOT.last_idx }
	for_loop_effect = {
		start = ROOT.last_idx
		end = tmp_pop
		set_variable = { pie_chart^v = 2 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	### IRL ###
	set_temp_variable = { tmp_pop = stream_irl }
	multiply_temp_variable = { tmp_pop = 1 }
	round_temp_variable = tmp_pop
	add_to_temp_variable = { tmp_pop = ROOT.last_idx }

	for_loop_effect = {
		start = ROOT.last_idx
		end = tmp_pop
		set_variable = { pie_chart^v = 3 }
		add_to_variable = { ROOT.last_idx = 1 }
	}

	### CREATIVE ###
	set_temp_variable = { tmp_pop = stream_creative }
	multiply_temp_variable = { tmp_pop = 1 }
	round_temp_variable = tmp_pop
	add_to_temp_variable = { tmp_pop = ROOT.last_idx }

	for_loop_effect = {
		start = ROOT.last_idx
		end = tmp_pop
		set_variable = { pie_chart^v = 4 }
	}

	clear_variable = ROOT.last_idx
	add_to_variable = { pie_chart_dirty = 1 }
}
