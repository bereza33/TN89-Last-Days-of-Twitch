enable_demob_economy_decisions = {
	if = {
		limit = {
			is_ai = no
			has_war = no
		}
		if = {
			limit = {
				has_unsupported_economic_law = yes
				NOT = {
					has_active_mission = demobilization_economic_2
				}	
			}
			activate_mission = demobilization_economic_2
		}
	}
}

trigger_strikes_escalation = {
	if = {
		limit = {
			NOT = {
				OR = {
					has_idea = strikes_1
					has_idea = strikes_2
					has_idea = strikes_3
				}
			}
		}
		add_ideas = strikes_1
		country_event = { id = stability.22 }
		set_country_flag = crisis_in_progress
	}
	else_if = {
		limit = {
			has_idea = strikes_1
		}
		swap_ideas = {
			remove_idea = strikes_1
			add_idea = strikes_2
		}
		country_event = { id = stability.22 }
	}
	else_if = {
		limit = {
			has_idea = strikes_2
		}
		swap_ideas = {
			remove_idea = strikes_2
			add_idea = strikes_3
		}
		country_event = { id = stability.22 }
	}
	else_if = {
		limit = {
			has_idea = strikes_3
			has_stability < 0.41
			NOT = {
				has_government = communism
				has_stability < 0.21
			}
		}
		add_popularity = { ideology = communism popularity = 0.05 }
		add_stability = -0.05
	}
	else_if = {
		limit = {
			has_idea = strikes_3
			has_stability < 0.21
			NOT = {
				has_government = communism
				communism > 0.25
			}
		}
		add_popularity = { ideology = communism popularity = 0.1 }
	}
	else_if = {
		limit = {
			has_idea = strikes_3
			has_stability < 0.41
			has_government = communism
			NOT = {
				has_stability < 0.21
			}
		}
		add_popularity = { ideology = democratic popularity = 0.05 }
		add_stability = -0.05
	}
	else_if = {
		limit = {
			has_idea = strikes_3
			has_stability < 0.21
			has_government = communism
			NOT = {
				democratic > 0.25
			}
		}
		add_popularity = { ideology = democratic popularity = 0.1 }
	}
	else_if = {
		limit = {
			has_idea = strikes_3
			has_stability < 0.21
			has_government = communism
			democratic > 0.25
		}
		start_civil_war = { ideology = democratic }
	}
	else_if = {
		limit = {
			has_idea = strikes_3
			has_stability < 0.21
			has_government = communism
			fascism > 0.25
		}
		start_civil_war = { ideology = fascism }
	}
	else_if = {
		limit = {
			has_idea = strikes_3
			has_stability < 0.21
			communism > 0.25
			NOT = {
				has_government = communism
			}
		}
		start_civil_war = { ideology = communism }
	}
}
